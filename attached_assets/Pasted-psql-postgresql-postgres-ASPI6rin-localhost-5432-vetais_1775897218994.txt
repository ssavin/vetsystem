WHERE table_name='file_clients' ORDER BY ordinal_position;

-- Совпадает ли client_id с kod_kado?
SELECT
  ah.client_id,
  fc_kod.kod_kado AS matched_by_kod,
  fc_id.id_kado   AS matched_by_id
FROM accounts_headers ah
LEFT JOIN file_clients fc_kod ON fc_kod.kod_kado::text = ah.client_id::text
LEFT JOIN file_clients fc_id  ON fc_id.id_kado::text   = ah.client_id::text
LIMIT 10;

-- Сколько счетов найдут владельца через kod_kado vs id_kado?
SELECT
  COUNT(*) FILTER (WHERE fc_kod.kod_kado IS NOT NULL) AS via_kod_kado,
  COUNT(*) FILTER (WHERE fc_id.id_kado IS NOT NULL)   AS via_id_kado,
  COUNT(*) AS total
FROM accounts_headers ah
LEFT JOIN file_clients fc_kod ON fc_kod.kod_kado::text = ah.client_id::text
LEFT JOIN file_clients fc_id  ON fc_id.id_kado::text   = ah.client_id::text
